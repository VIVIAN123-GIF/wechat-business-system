```cpp
// promotion/IPromotion.h
#pragma once
#include <string>
#include <memory>

class IPromotion {
public:
    virtual ~IPromotion() = default;
    virtual double apply(double subtotal) = 0;          // 应用促销，返回优惠后的金额
    virtual double getDiscountAmount(double subtotal) = 0; // 返回优惠了多少
    virtual std::string getDescription() const = 0;
    virtual int getId() const = 0;
};

// promotion/DiscountPromotion.h
class DiscountPromotion : public IPromotion {
private:
    int id;
    double rate;  // 0.85 表示85折
public:
    DiscountPromotion(int id, double r) : id(id), rate(r) {}
    double apply(double subtotal) override { return subtotal * rate; }
    double getDiscountAmount(double subtotal) override { return subtotal * (1 - rate); }
    std::string getDescription() const override { 
        return std::to_string((int)(rate * 100)) + "%折扣"; 
    }
    int getId() const override { return id; }
};

// promotion/CouponPromotion.h (满减券)
class CouponPromotion : public IPromotion {
private:
    int id;
    double threshold;
    double reduce;
public:
    CouponPromotion(int id, double t, double r) : id(id), threshold(t), reduce(r) {}
    double apply(double subtotal) override {
        return (subtotal >= threshold) ? (subtotal - reduce) : subtotal;
    }
    double getDiscountAmount(double subtotal) override {
        return (subtotal >= threshold) ? reduce : 0;
    }
    std::string getDescription() const override {
        return "满" + std::to_string((int)threshold) + "减" + std::to_string((int)reduce);
    }
    int getId() const override { return id; }
};

// promotion/PromotionDecorator.h —— 装饰器基类
class PromotionDecorator : public IPromotion {
protected:
    std::unique_ptr<IPromotion> wrapped;
public:
    explicit PromotionDecorator(std::unique_ptr<IPromotion> p) : wrapped(std::move(p)) {}
    double apply(double subtotal) override {
        return wrapped ? wrapped->apply(subtotal) : subtotal;
    }
    double getDiscountAmount(double subtotal) override {
        return wrapped ? wrapped->getDiscountAmount(subtotal) : 0;
    }
    std::string getDescription() const override {
        return wrapped ? wrapped->getDescription() : "";
    }
    int getId() const override { return wrapped ? wrapped->getId() : -1; }
};

// 具体装饰器：叠加额外折扣（比如会员折上折）
class ExtraDiscountDecorator : public PromotionDecorator {
private:
    double extraRate;
public:
    ExtraDiscountDecorator(std::unique_ptr<IPromotion> p, double rate)
        : PromotionDecorator(std::move(p)), extraRate(rate) {}
    double apply(double subtotal) override {
        double base = PromotionDecorator::apply(subtotal);
        return base * extraRate;
    }
    double getDiscountAmount(double subtotal) override {
        double baseAfterWrapped = wrapped ? wrapped->apply(subtotal) : subtotal;
        double final = baseAfterWrapped * extraRate;
        return subtotal - final;
    }
    std::string getDescription() const override {
        return PromotionDecorator::getDescription() + " + 额外" + 
               std::to_string((int)(extraRate * 100)) + "%折扣";
    }
};
```

```cpp
// server/handler/CheckoutHandler.h
class CheckoutHandler : public ICommandHandler {
private:
    std::shared_ptr<ProductRepository> productRepo;
    std::shared_ptr<OrderRepository> orderRepo;
    std::shared_ptr<PromotionFactory> promoFactory;
    std::mutex& stockMutex;  // 全局库存互斥锁（引用）
public:
    CheckoutHandler(std::shared_ptr<ProductRepository> pr,
                    std::shared_ptr<OrderRepository> or,
                    std::shared_ptr<PromotionFactory> pf,
                    std::mutex& mutex)
        : productRepo(pr), orderRepo(or), promoFactory(pf), stockMutex(mutex) {}

    bool execute(uint32_t requestCode, const std::string& jsonReq, 
                 std::string& jsonResp) override {
        // 1. 解析请求
        auto j = nlohmann::json::parse(jsonReq);
        int userId = j["user_id"];
        auto itemsJson = j["items"];  // 数组
        auto promoIds = j["promo_ids"]; // 数组

        // 2. 构建促销链（装饰模式）
        std::unique_ptr<IPromotion> promoChain = nullptr;
        for (int pid : promoIds) {
            auto promo = promoFactory->create(pid);
            if (promo) {
                if (!promoChain) {
                    promoChain = std::move(promo);
                } else {
                    // 用装饰器包装已有链
                    promoChain = std::make_unique<ExtraDiscountDecorator>(
                        std::move(promoChain), 
                        dynamic_cast<DiscountPromotion*>(promo.get())->getRate() // 简化，实际需更通用
                    );
                }
            }
        }

        // 3. 计算原价 & 锁定库存（加锁）
        double total = 0;
        std::lock_guard<std::mutex> lock(stockMutex);
        for (auto& item : itemsJson) {
            int pid = item["product_id"];
            int qty = item["quantity"];
            auto prod = productRepo->findById(pid);
            if (!prod || prod->stock < qty) {
                jsonResp = R"({"err_code":1001,"err_msg":"库存不足"})";
                return false;
            }
            total += prod->price * qty;
            // 扣减库存（内存中扣减，稍后持久化）
            prod->stock -= qty;
            productRepo->update(*prod);
        }

        // 4. 应用促销链
        double finalAmount = total;
        double discountTotal = 0;
        if (promoChain) {
            finalAmount = promoChain->apply(total);
            discountTotal = total - finalAmount;
        }

        // 5. 生成订单并保存
        Order order;
        order.userId = userId;
        order.totalAmount = total;
        order.discountAmount = discountTotal;
        order.finalAmount = finalAmount;
        order.status = 1; // 已支付
        order.createTime = getCurrentTime();
        if (!orderRepo->insert(order)) {
            jsonResp = R"({"err_code":1009,"err_msg":"订单保存失败"})";
            return false;
        }

        // 6. 返回成功响应
        nlohmann::json resp;
        resp["order_id"] = order.id;
        resp["order_no"] = order.orderNo;
        resp["original"] = total;
        resp["discount"] = discountTotal;
        resp["final"] = finalAmount;
        jsonResp = resp.dump();
        return true;
    }
};
```

```cpp
bool TcpServer::start() {
    WSADATA wsaData;
    if (WSAStartup(MAKEWORD(2,2), &wsaData) != 0) {
        std::cerr << "WSAStartup failed" << std::endl;
        return false;
    }

    listenFd = socket(AF_INET, SOCK_STREAM, 0);
    if (listenFd == INVALID_SOCKET) {
        std::cerr << "socket() failed" << std::endl;
        WSACleanup();
        return false;
    }

    sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(port);
    if (bind(listenFd, (sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR) {
        std::cerr << "bind() failed" << std::endl;
        closesocket(listenFd);
        WSACleanup();
        return false;
    }

    if (listen(listenFd, SOMAXCONN) == SOCKET_ERROR) {
        std::cerr << "listen() failed" << std::endl;
        closesocket(listenFd);
        WSACleanup();
        return false;
    }

    running = true;
    // 启动接受线程
    acceptThread = std::thread(&TcpServer::acceptLoop, this);
    // 启动工作线程池
    workerPool.start();
    return true;
}

void TcpServer::acceptLoop() {
    while (running) {
        SOCKET clientFd = accept(listenFd, nullptr, nullptr);
        if (clientFd == INVALID_SOCKET) {
            if (running) std::cerr << "accept() failed" << std::endl;
            continue;
        }
        // 为每个客户端启动一个处理线程
        std::thread(&TcpServer::clientHandler, this, clientFd).detach();
    }
}

void TcpServer::clientHandler(SOCKET clientFd) {
    while (running) {
        ProtocolHeader header;
        std::string body;
        if (!recvFullPacket(clientFd, header, body)) {
            break;  // 客户端断开或错误
        }
        // 构造Task并入队
        Task task;
        task.clientFd = clientFd;
        task.protocolCode = header.code;
        task.jsonBody = body;
        TaskQueue::getInstance().push(task);
    }
    closesocket(clientFd);
}
```

```cpp
// common/ProtocolEncoder.cpp
std::vector<char> ProtocolEncoder::encode(uint32_t code, const std::string& body) {
    ProtocolHeader header;
    header.magic = 0x4D495F53;  // "MI_S"
    header.version = 0x00000001;
    header.code = htonl(code);   // 网络字节序
    header.length = htonl(body.size());

    std::vector<char> buffer(sizeof(header) + body.size());
    memcpy(buffer.data(), &header, sizeof(header));
    memcpy(buffer.data() + sizeof(header), body.data(), body.size());
    return buffer;
}

bool ProtocolEncoder::decode(const std::vector<char>& buffer, 
                             uint32_t& outCode, std::string& outBody) {
    if (buffer.size() < sizeof(ProtocolHeader)) return false;
    ProtocolHeader header;
    memcpy(&header, buffer.data(), sizeof(header));
    if (header.magic != 0x4D495F53) return false;
    outCode = ntohl(header.code);
    uint32_t len = ntohl(header.length);
    if (buffer.size() < sizeof(header) + len) return false;
    outBody.assign(buffer.data() + sizeof(header), len);
    return true;
}

// 接收完整包（处理粘包）
bool recvFullPacket(SOCKET fd, ProtocolHeader& header, std::string& body) {
    char headBuf[sizeof(ProtocolHeader)];
    int received = 0;
    while (received < sizeof(ProtocolHeader)) {
        int ret = recv(fd, headBuf + received, sizeof(ProtocolHeader) - received, 0);
        if (ret <= 0) return false;
        received += ret;
    }
    memcpy(&header, headBuf, sizeof(header));
    if (header.magic != 0x4D495F53) return false;
    uint32_t bodyLen = ntohl(header.length);
    body.resize(bodyLen);
    received = 0;
    while (received < bodyLen) {
        int ret = recv(fd, body.data() + received, bodyLen - received, 0);
        if (ret <= 0) return false;
        received += ret;
    }
    return true;
}
```

```cpp
// server/TaskQueue.cpp (单例)
void TaskQueue::push(const Task& task) {
    std::lock_guard<std::mutex> lock(mtx);
    queue.push(task);
    cond.notify_one();  // 唤醒一个等待的Worker
}

Task TaskQueue::pop() {
    std::unique_lock<std::mutex> lock(mtx);
    cond.wait(lock, [this](){ return !queue.empty() || shutdown; });
    if (shutdown && queue.empty()) {
        Task empty; empty.protocolCode = 0;
        return empty;
    }
    Task task = queue.front();
    queue.pop();
    return task;
}

// server/WorkerPool.cpp
void WorkerPool::workerLoop() {
    while (running) {
        Task task = TaskQueue::getInstance().pop();
        if (task.protocolCode == 0) continue;
        handleTask(task);
    }
}

void WorkerPool::handleTask(const Task& task) {
    auto handler = HandlerFactory::createHandler(task.protocolCode, 
                                                 productRepo, orderRepo, 
                                                 promoFactory, stockMutex);
    if (!handler) {
        // 未知协议码，返回错误
        sendError(task.clientFd, 4001, "Unknown protocol");
        return;
    }
    std::string jsonResp;
    bool success = handler->execute(task.protocolCode, task.jsonBody, jsonResp);
    // 发送响应
    uint32_t respCode = (success ? getRespCode(task.protocolCode) : 4001);
    auto buffer = ProtocolEncoder::encode(respCode, jsonResp);
    send(task.clientFd, buffer.data(), buffer.size(), 0);
}
```

```cpp
bool TcpClient::receiveResponse(int& outCode, std::string& outBody, int timeoutSec) {
    // 设置超时
```

```cpp
class SQLiteHelper {
private:
    sqlite3* db;
public:
    SQLiteHelper(const std::string& dbFile) {
        if (sqlite3_open(dbFile.c_str(), &db) != SQLITE_OK) {
            throw std::runtime_error("Can't open database");
        }
    }
    ~SQLiteHelper() { if (db) sqlite3_close(db); }

    // 执行非查询（INSERT/UPDATE/DELETE）
    bool execute(const std::string& sql) {
        char* errMsg = nullptr;
        int rc = sqlite3_exec(db, sql.c_str(), nullptr, nullptr, &errMsg);
        if (rc != SQLITE_OK) {
            std::cerr << "SQL error: " << errMsg << std::endl;
            sqlite3_free(errMsg);
            return false;
        }
        return true;
    }

    // 执行查询，通过回调或迭代器获取结果
    sqlite3_stmt* prepare(const std::string& sql) {
        sqlite3_stmt* stmt;
        if (sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt, nullptr) != SQLITE_OK) {
            return nullptr;
        }
        return stmt;
    }
};
```

```cpp
std::vector<Product> ProductRepository::findAll() {
    std::vector<Product> result;
    std::string sql = "SELECT id, name, price, stock, description, category FROM products WHERE status=1";
    sqlite3_stmt* stmt = dbHelper->prepare(sql);
    if (!stmt) return result;
    while (sqlite3_step(stmt) == SQLITE_ROW) {
        Product p;
        p.id = sqlite3_column_int(stmt, 0);
        p.name = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
        p.price = sqlite3_column_double(stmt, 2);
        p.stock = sqlite3_column_int(stmt, 3);
        if (sqlite3_column_type(stmt, 4) != SQLITE_NULL)
            p.description = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 4));
        if (sqlite3_column_type(stmt, 5) != SQLITE_NULL)
            p.category = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 5));
        result.push_back(p);
    }
    sqlite3_finalize(stmt);
    return result;
}
```

### 二、遇见了哪些技术问题，如何解决的？

本周编码过程中遇到了若干个实际技术问题，现逐一列出并给出解决方案。

#### 问题一：Winsock头文件包含顺序导致编译错误

**现象**：在`TcpServer.cpp`中`#include <winsock2.h>`后，编译报错`error C2011: 'sockaddr' : 'struct' type redefinition`。

**原因**：`windows.h`默认包含了`winsock.h`（旧版），与`winsock2.h`冲突。

**解决方法**：
- 在所有源文件的最前面**先** `#include <winsock2.h>`，再`#include <windows.h>`。
- 或定义宏`WIN32_LEAN_AND_MEAN`阻止`windows.h`包含旧版socket。
- 我选择了在预编译头中加入`#define WIN32_LEAN_AND_MEAN`并调整头文件顺序。

#### 问题二：TCP发送/接收时遇到“粘包”导致协议解析错乱

**现象**：客户端连续发送两个请求，服务端一次`recv`接收到两个包的数据，解析时出错。

**原因**：TCP是流式传输，没有消息边界。

**解决方法**：
- 按照预定设计，采用**定长头部+变长体部**。
- 服务端先读8字节头部，解析出`length`字段，再循环读`length`字节。即使一次`recv`收到多个包，也会严格按照长度切割。
- 编写了`recvFullPacket`函数（见上文），经过测试已能正确拆分。

#### 问题三：SQLite插入订单时外键约束失败

**现象**：插入`orders`表时，`user_id`在`users`表中存在，但依然报`FOREIGN KEY constraint failed`。

**原因**：SQLite默认关闭外键约束支持。

**解决方法**：
- 每次打开数据库后执行`PRAGMA foreign_keys = ON;`。
- 将`sqlite3_open`后立即执行`dbHelper->execute("PRAGMA foreign_keys = ON;")`。
- 之后外键约束正常工作。

#### 问题四：多线程环境下，`std::cout`输出混乱

**现象**：服务端多个工作线程同时输出日志，终端显示交错字符。

**原因**：`std::cout`不是线程安全的，多个线程同时写入导致竞态。

**解决方法**：
- 引入**日志队列**，所有日志不直接`cout`，而是放入`LogQueue`，由一个单独的日志线程异步输出。
- 或简单地在每个输出前加`std::lock_guard<std::mutex>`保护`std::cout`。
- 我选择封装一个`Logger`单例，内部维护`mutex`，所有日志通过`Logger::info()`输出，加锁保证原子性。

```cpp
class Logger {
public:
    static Logger& getInstance() { static Logger instance; return instance; }
    void info(const std::string& msg) {
        std::lock_guard<std::mutex> lock(mtx);
        std::cout << "[INFO] " << msg << std::endl;
    }
    void error(const std::string& msg) {
        std::lock_guard<std::mutex> lock(mtx);
        std::cerr << "[ERROR] " << msg << std::endl;
    }
private:
    std::mutex mtx;
};
```

#### 问题五：客户端接收异步推送消息时阻塞在主菜单循环

**现象**：客户端发起结算（1030）后，服务端立即返回2030（处理中），随后异步推送2031。但客户端的`receiveResponse`是阻塞的，且只在用户操作时调用，无法主动接收异步推送。

**解决方法**：
- 客户端增加**接收线程**，专门阻塞`recv`。
- 主线程与接收线程通过`std::queue`传递消息。
- 主循环每次迭代检查队列，若有异步消息则显示。
```cpp
// client/main.cpp 伪代码
std::thread receiver([&](){
    while (connected) {
        int code; std::string body;
        if (tcpClient->receiveResponse(code, body, 1)) { // 1秒超时
            std::lock_guard<std::mutex> lock(msgMutex);
            msgQueue.push({code, body});
        }
    }
});
receiver.detach();

// 主循环中检查消息队列
while (running) {
    if (!msgQueue.empty()) {
        auto msg = popMsg();
        if (msg.code == 2031) renderer->drawOrderResult(msg.body);
    }
    // 显示菜单...
}
```

**困难一：并发场景下的数据库事务**

尽管使用了内存锁和SQL原子更新，但在高并发下（多个线程同时扣减不同商品的库存），如何保证**一组操作**（查库存→扣库存→生成订单→插入订单明细）的ACID特性？目前每个操作单独执行SQL，如果中间步骤失败，会导致数据不一致。

**应对策略**：下周计划引入SQLite的**事务**机制，使用`BEGIN TRANSACTION`和`COMMIT`/`ROLLBACK`包裹整个结算流程。如果任何一步失败，全部回滚。

**应对策略**：考虑使用`select`或`WSAEventSelect`实现事件驱动，避免轮询。但目前先保证功能正确，性能优化留待最后一周。

**困难三：装饰模式的动态组合与JSON序列化**

促销链的组装需要根据客户端传来的`promo_ids`动态构建。但装饰器之间互相嵌套，序列化时如何还原完整的促销链描述？当前我直接在`CheckoutHandler`中硬编码构建逻辑，灵活性不足。
